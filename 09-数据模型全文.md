# 09 数据模型全文

以下全部模型直接来自 1.13 OpenAPI，保留 required、默认/枚举、边界、oneOf、引用及扩展；不省略嵌套字段。HTTP操作中的同名模型对应本章。候选ASR模型单独提供，不混写进主基线。

## SIPConfig



```json
{
  "type": "object",
  "properties": {
    "listen": {
      "type": "string",
      "description": "本地 UDP 监听地址与端口。",
      "minLength": 1,
      "pattern": "^[0-9.]+:[0-9]+$"
    },
    "advertise": {
      "type": "string",
      "description": "向对端宣告的 SIP 地址与端口。",
      "minLength": 1,
      "pattern": "^[0-9.]+:[0-9]+$"
    },
    "upstream": {
      "type": "string",
      "description": "固定 B 腿上游地址与端口。",
      "minLength": 1,
      "pattern": "^[0-9.]+:[0-9]+$"
    },
    "trusted_networks": {
      "type": "array",
      "items": {
        "type": "string",
        "description": ""
      },
      "description": "允许初始信令来源的 IPv4 CIDR 列表。",
      "minItems": 1
    },
    "setup_timeout_ms": {
      "type": "integer",
      "description": "呼叫建立阶段超时，毫秒。",
      "minimum": 1000,
      "maximum": 180000
    },
    "ack_timeout_ms": {
      "type": "integer",
      "description": "最终响应等待 ACK 的最长时间，毫秒。",
      "minimum": 1000,
      "maximum": 64000
    },
    "max_call_seconds": {
      "type": "integer",
      "description": "成功 ACK 后允许的最大通话时长，秒。",
      "minimum": 1,
      "maximum": 86400
    },
    "stream": {
      "$ref": "#/components/schemas/SIPStreamConfig"
    },
    "trunk_auth": {
      "$ref": "#/components/schemas/SIPTrunkAuth"
    },
    "registration": {
      "$ref": "#/components/schemas/SIPRegistration"
    },
    "local_extensions": {
      "type": "array",
      "maxItems": 256,
      "uniqueItems": true,
      "items": {
        "type": "string",
        "pattern": "^[+0-9]{1,32}$"
      },
      "description": "未配置sip.dialplan时使用的精确本地接听号码；启用XML后此列表不作为未匹配回退。"
    },
    "dialplan": {
      "oneOf": [
        {
          "$ref": "#/components/schemas/SIPDialplan"
        },
        {
          "type": "null"
        }
      ],
      "description": "可选启动级本地XML拨号计划。文件与context由部署配置管理，页面只读。"
    }
  },
  "description": "",
  "required": [
    "listen",
    "advertise",
    "upstream",
    "trusted_networks",
    "setup_timeout_ms",
    "ack_timeout_ms",
    "max_call_seconds"
  ],
  "additionalProperties": false
}
```

## Config

```json
{
  "type": "object",
  "properties": {
    "sip": {
      "$ref": "#/components/schemas/SIPConfig",
      "description": "信令地址、信任与定时器。"
    },
    "media": {
      "$ref": "#/components/schemas/MediaConfig",
      "description": "媒体分片与 socket 配置。"
    },
    "limits": {
      "$ref": "#/components/schemas/Limits",
      "description": "本次启动硬容量边界。"
    },
    "admin": {
      "$ref": "#/components/schemas/AdminListenConfig",
      "description": "本地管理入口。"
    },
    "journal": {
      "$ref": "#/components/schemas/JournalConfig",
      "description": "业务日志。"
    },
    "pcm_stream": {
      "$ref": "#/components/schemas/PCMStreamConfig",
```
